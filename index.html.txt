/* =========================
   CRIAR DADO D6
========================= */

function criarDado(numero) {

    const dado =
        document.createElement("div");

    dado.className =
        "dado-visual";


    let pontos = [];


    if (numero === 1) {

        pontos = [
            ["center", true]
        ];

    }


    if (numero === 2) {

        pontos = [
            ["top-left", false],
            ["bottom-right", false]
        ];

    }


    if (numero === 3) {

        pontos = [
            ["top-left", false],
            ["center", false],
            ["bottom-right", false]
        ];

    }


    if (numero === 4) {

        pontos = [
            ["top-left", false],
            ["top-right", false],
            ["bottom-left", false],
            ["bottom-right", false]
        ];

    }


    if (numero === 5) {

        pontos = [
            ["top-left", false],
            ["top-right", false],
            ["center", false],
            ["bottom-left", false],
            ["bottom-right", false]
        ];

    }


    if (numero === 6) {

        pontos = [
            ["top-left", false],
            ["top-right", false],
            ["middle-left", false],
            ["middle-right", false],
            ["bottom-left", false],
            ["bottom-right", false]
        ];

    }


    for (let pontoInfo of pontos) {

        const ponto =
            document.createElement("span");


        ponto.className =
            "ponto " + pontoInfo[0];


        if (pontoInfo[1] === true) {

            ponto.classList.add(
                "vermelho"
            );

        }


        dado.appendChild(
            ponto
        );

    }


    return dado;

}



/* =========================
   MOSTRAR DADOS
========================= */

function mostrarDados(
    resultados,
    tipo
) {

    const area =
        document.getElementById(
            "areaDado"
        );


    area.innerHTML = "";


    if (resultados.length >= 4) {

        area.classList.add(
            "muitos"
        );

    }

    else {

        area.classList.remove(
            "muitos"
        );

    }


    for (let numero of resultados) {


        if (tipo === 6) {

            const dado =
                criarDado(numero);


            area.appendChild(
                dado
            );

        }


        else {

            const dado =
                document.createElement(
                    "div"
                );


            dado.className =
                "numero-dado";


            dado.textContent =
                numero;


            area.appendChild(
                dado
            );

        }

    }

}



/* =========================
   SOM
========================= */

const somDado = new Audio("PASTA/SEU-SOM.mp3");

function tocarSomDado() {

    somDado.currentTime = 0;
    somDado.play();

}



/* =========================
   JOGAR
========================= */

function jogarDado() {


    const quantidade =
        Number(
            document.getElementById(
                "quantidade"
            ).value
        );


    const tipo =
        Number(
            document.getElementById(
                "tipo"
            ).value
        );


    const area =
        document.getElementById(
            "areaDado"
        );


    /* SOM */

    tocarSomDado();


    /* ANIMAÇÃO */

    const animacao =
        setInterval(function() {


            let temporarios = [];


            for (
                let i = 0;
                i < quantidade;
                i++
            ) {


                temporarios.push(

                    Math.floor(
                        Math.random() *
                        tipo
                    ) + 1

                );

            }


            mostrarDados(
                temporarios,
                tipo
            );



            const dados =
                area.children;


            for (
                let dado of dados
            ) {


                dado.classList.add(
                    "rolando"
                );

            }


        }, 100);



    /* RESULTADO FINAL */

    setTimeout(function() {


        clearInterval(
            animacao
        );


        let resultados = [];

        let total = 0;


        for (
            let i = 0;
            i < quantidade;
            i++
        ) {


            const numero =

                Math.floor(
                    Math.random() *
                    tipo
                ) + 1;


            resultados.push(
                numero
            );


            total +=
                numero;

        }



        mostrarDados(
            resultados,
            tipo
        );



        document.getElementById(
            "resultado"
        ).textContent =

            "Resultados: " +
            resultados.join(", ");



        document.getElementById(
            "total"
        ).textContent =

            "Total: " +
            total;



        adicionarHistorico(
            tipo,
            quantidade,
            resultados,
            total
        );


    }, 900);

}



/* =========================
   HISTÓRICO
========================= */

function adicionarHistorico(
    tipo,
    quantidade,
    resultados,
    total
) {


    const lista =
        document.getElementById(
            "listaHistorico"
        );


    const vazio =
        lista.querySelector(
            ".vazio"
        );


    if (vazio) {

        vazio.remove();

    }



    const jogada =
        document.createElement(
            "p"
        );


    jogada.textContent =

        "D" + tipo +

        " × " + quantidade +

        " → " +

        resultados.join(", ") +

        " | Total: " +

        total;



    lista.prepend(
        jogada
    );



    /* MÁXIMO 10 JOGADAS */

    while (
        lista.children.length > 10
    ) {


        lista.removeChild(
            lista.lastElementChild
        );

    }

}



/* =========================
   LIMPAR HISTÓRICO
========================= */

function limparHistorico() {


    const lista =
        document.getElementById(
            "listaHistorico"
        );


    lista.innerHTML =

        '<p class="vazio">' +
        'Nenhuma jogada ainda.' +
        '</p>';

}
